Практическая работа 10. Код для терминала Linux.
Файл в кодировке UTF-8.

Как вставить:
1. В Linux перейди в пустую папку, например: mkdir -p ~/pr10 && cd ~/pr10
2. Копируй целиком блок от cat до EOF включительно и вставляй в терминал.
3. Строки из знаков «=» не копируй: это только подписи.
4. После каждого файла — команды сборки из следующего блока.

Чем код отличается от картинок в методичке:
- symlink_depth сам создаёт каталог links и работает внутри него.
  В тексте задания mkdir ошибочно отнесена к fcntl.h, объявление в sys/stat.h.
- myls подключает limits.h: там объявлен PATH_MAX.
- кода ls -al в методичке нет. Его делает myls_al.c, рядом лежит README.

Перед повторным запуском symlink_depth удали старый каталог: rm -rf links
========== symlink_depth.c ==========

cat > symlink_depth.c << 'EOF'
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <sys/stat.h>
#include <string.h>

int main(void) {
    const char *dir = "links";

    /* В методичке файлы создаются в текущем каталоге.
       Задание просит отдельную директорию, поэтому создаём её здесь.
       mkdir объявлена в <sys/stat.h>, а не в <fcntl.h>. */
    if (mkdir(dir, 0755) < 0 && errno != EEXIST) {
        perror("Ошибка при создании каталога");
        return 1;
    }
    if (chdir(dir) < 0) {
        perror("Ошибка при переходе в каталог");
        return 1;
    }

    const char *filename = "a";
    int fd = creat(filename, 0666);
    if (fd < 0) {
        perror("Ошибка при создании файла");
        return 1;
    }
    close(fd);

    char prev_link[256] = "a";
    char new_link[256];
    int depth = 0;

    while (1) {
        snprintf(new_link, sizeof(new_link), "a%c", 'a' + depth);

        if (symlink(prev_link, new_link) < 0) {
            perror("Ошибка при создании символической ссылки");
            break;
        }

        fd = open(new_link, O_RDONLY);
        if (fd < 0) {
            if (errno == ELOOP) {
                printf("Достигнута максимальная глубина рекурсии: %d\n", depth);
                break;
            }
            perror("Ошибка при открытии файла");
            break;
        }

        close(fd);
        snprintf(prev_link, sizeof(prev_link), "%s", new_link);
        depth++;
    }

    return 0;
}
EOF

========== myls.c ==========

cat > myls.c << 'EOF'
#include <stdio.h>
#include <stdlib.h>
#include <dirent.h>
#include <sys/stat.h>
#include <string.h>
#include <unistd.h>
#include <errno.h>
#include <limits.h>

void print_file_type(const struct stat *st) {
    if (S_ISREG(st->st_mode)) {
        printf(" - Regular file\n");
    } else if (S_ISDIR(st->st_mode)) {
        printf(" - Directory\n");
    } else if (S_ISLNK(st->st_mode)) {
        printf(" - Symbolic link\n");
    } else if (S_ISCHR(st->st_mode)) {
        printf(" - Character device\n");
    } else if (S_ISBLK(st->st_mode)) {
        printf(" - Block device\n");
    } else if (S_ISFIFO(st->st_mode)) {
        printf(" - FIFO (named pipe)\n");
    } else if (S_ISSOCK(st->st_mode)) {
        printf(" - Socket\n");
    } else {
        printf(" - Unknown file type\n");
    }
}

void list_directory(const char *path) {
    DIR *dir = opendir(path);
    if (!dir) {
        perror("opendir");
        return;
    }

    struct dirent *entry;
    char fullpath[PATH_MAX];

    while ((entry = readdir(dir)) != NULL) {
        snprintf(fullpath, sizeof(fullpath), "%s/%s", path, entry->d_name);

        struct stat st;
        if (lstat(fullpath, &st) == -1) {
            perror("lstat");
            continue;
        }

        printf("%s", entry->d_name);
        print_file_type(&st);

        if (S_ISLNK(st.st_mode)) {
            char link_target[PATH_MAX];
            ssize_t len = readlink(fullpath, link_target, sizeof(link_target) - 1);
            if (len != -1) {
                link_target[len] = '\0';
                printf(" -> %s\n", link_target);
            } else {
                perror("readlink");
            }
        }
    }

    closedir(dir);
}

int main(int argc, char *argv[]) {
    const char *path = (argc > 1) ? argv[1] : ".";
    list_directory(path);
    return 0;
}
EOF

========== myls_al.c ==========

cat > myls_al.c << 'EOF'
#define _DEFAULT_SOURCE

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <sys/stat.h>
#include <sys/sysmacros.h>
#include <unistd.h>
#include <errno.h>
#include <limits.h>
#include <pwd.h>
#include <grp.h>
#include <time.h>
#include <locale.h>

typedef struct {
    char *name;
    struct stat st;
    int ok;
    char target[PATH_MAX];
} Entry;

static int cmp_entry(const void *a, const void *b) {
    const Entry *ea = a;
    const Entry *eb = b;
    return strcoll(ea->name, eb->name);
}

static void mode_string(mode_t mode, char out[11]) {
    if (S_ISREG(mode)) {
        out[0] = '-';
    } else if (S_ISDIR(mode)) {
        out[0] = 'd';
    } else if (S_ISLNK(mode)) {
        out[0] = 'l';
    } else if (S_ISCHR(mode)) {
        out[0] = 'c';
    } else if (S_ISBLK(mode)) {
        out[0] = 'b';
    } else if (S_ISFIFO(mode)) {
        out[0] = 'p';
    } else if (S_ISSOCK(mode)) {
        out[0] = 's';
    } else {
        out[0] = '?';
    }

    out[1] = (mode & S_IRUSR) ? 'r' : '-';
    out[2] = (mode & S_IWUSR) ? 'w' : '-';
    if (mode & S_ISUID) {
        out[3] = (mode & S_IXUSR) ? 's' : 'S';
    } else {
        out[3] = (mode & S_IXUSR) ? 'x' : '-';
    }

    out[4] = (mode & S_IRGRP) ? 'r' : '-';
    out[5] = (mode & S_IWGRP) ? 'w' : '-';
    if (mode & S_ISGID) {
        out[6] = (mode & S_IXGRP) ? 's' : 'S';
    } else {
        out[6] = (mode & S_IXGRP) ? 'x' : '-';
    }

    out[7] = (mode & S_IROTH) ? 'r' : '-';
    out[8] = (mode & S_IWOTH) ? 'w' : '-';
    if (mode & S_ISVTX) {
        out[9] = (mode & S_IXOTH) ? 't' : 'T';
    } else {
        out[9] = (mode & S_IXOTH) ? 'x' : '-';
    }
    out[10] = '\0';
}

static void join_path(char *dst, size_t n, const char *dir, const char *name) {
    size_t len = strlen(dir);
    if (strcmp(dir, "/") == 0) {
        snprintf(dst, n, "/%s", name);
    } else if (len > 0 && dir[len - 1] == '/') {
        snprintf(dst, n, "%s%s", dir, name);
    } else {
        snprintf(dst, n, "%s/%s", dir, name);
    }
}

static void format_time(time_t mtime, char *buf, size_t n) {
    time_t now = time(NULL);
    struct tm *tm = localtime(&mtime);
    if (!tm) {
        snprintf(buf, n, "?");
        return;
    }
    double diff = difftime(now, mtime);
    if (diff > 15552000 || diff < -3600) {
        strftime(buf, n, "%b %e  %Y", tm);
    } else {
        strftime(buf, n, "%b %e %H:%M", tm);
    }
}

int main(int argc, char *argv[]) {
    const char *path = (argc > 1) ? argv[1] : ".";
    DIR *dir;
    struct dirent *de;
    Entry *items = NULL;
    size_t count = 0;
    size_t cap = 0;
    long total_blocks = 0;
    size_t i;

    setlocale(LC_ALL, "");

    dir = opendir(path);
    if (!dir) {
        perror("opendir");
        return 1;
    }

    while ((de = readdir(dir)) != NULL) {
        Entry *tmp;
        if (count == cap) {
            cap = cap ? cap * 2 : 64;
            tmp = realloc(items, cap * sizeof(*items));
            if (!tmp) {
                perror("realloc");
                closedir(dir);
                return 1;
            }
            items = tmp;
        }
        items[count].name = strdup(de->d_name);
        if (!items[count].name) {
            perror("strdup");
            closedir(dir);
            return 1;
        }
        items[count].ok = 0;
        items[count].target[0] = '\0';
        count++;
    }
    closedir(dir);

    qsort(items, count, sizeof(*items), cmp_entry);

    for (i = 0; i < count; i++) {
        char full[PATH_MAX];
        join_path(full, sizeof(full), path, items[i].name);
        if (lstat(full, &items[i].st) == -1) {
            perror(full);
            continue;
        }
        items[i].ok = 1;
        total_blocks += (long)items[i].st.st_blocks;
        if (S_ISLNK(items[i].st.st_mode)) {
            ssize_t len = readlink(full, items[i].target, sizeof(items[i].target) - 1);
            if (len >= 0) {
                items[i].target[len] = '\0';
            } else {
                perror("readlink");
                items[i].target[0] = '\0';
            }
        }
    }

    /* st_blocks считается в блоках по 512 байт, ls по умолчанию печатает килобайты */
    printf("total %ld\n", total_blocks / 2);

    for (i = 0; i < count; i++) {
        char mode[11];
        char size_buf[32];
        char time_buf[64];
        char uid_buf[32];
        char gid_buf[32];
        const char *owner;
        const char *group;
        struct passwd *pw;
        struct group *gr;

        if (!items[i].ok) {
            free(items[i].name);
            continue;
        }

        mode_string(items[i].st.st_mode, mode);
        pw = getpwuid(items[i].st.st_uid);
        gr = getgrgid(items[i].st.st_gid);
        if (pw) {
            owner = pw->pw_name;
        } else {
            snprintf(uid_buf, sizeof(uid_buf), "%u", (unsigned)items[i].st.st_uid);
            owner = uid_buf;
        }
        if (gr) {
            group = gr->gr_name;
        } else {
            snprintf(gid_buf, sizeof(gid_buf), "%u", (unsigned)items[i].st.st_gid);
            group = gid_buf;
        }

        if (S_ISCHR(items[i].st.st_mode) || S_ISBLK(items[i].st.st_mode)) {
            snprintf(size_buf, sizeof(size_buf), "%u, %u",
                     major(items[i].st.st_rdev), minor(items[i].st.st_rdev));
        } else {
            snprintf(size_buf, sizeof(size_buf), "%lld",
                     (long long)items[i].st.st_size);
        }

        format_time(items[i].st.st_mtime, time_buf, sizeof(time_buf));

        printf("%s %3lu %-8s %-8s %8s %s %s",
               mode,
               (unsigned long)items[i].st.st_nlink,
               owner,
               group,
               size_buf,
               time_buf,
               items[i].name);
        if (S_ISLNK(items[i].st.st_mode) && items[i].target[0] != '\0') {
            printf(" -> %s", items[i].target);
        }
        printf("\n");
        free(items[i].name);
    }

    free(items);
    return 0;
}
EOF

========== README ==========

cat > README << 'EOF'
Аналог команд ls и ls -al

myls
    Печатает имя и тип каждого файла в каталоге.
    Для символьной ссылки следующей строкой печатает цель (readlink).
    Тип определяется по lstat, поэтому ссылка не подменяется файлом, на который она указывает.

myls_al
    Приближает вывод к ls -al: строка total, тип и права, число жёстких ссылок,
    владелец, группа, размер, время изменения и имя.
    Для символьной ссылки в той же строке печатается «-> цель».
    Для устройства вместо размера печатаются старший и младший номера.
    Имена выводятся в алфавитном порядке, включая . и ..

Сборка:
    gcc -o myls myls.c
    gcc -o myls_al myls_al.c

Запуск:
    ./myls /dev
    ./myls_al /dev
    ./myls_al

Если путь не передан, обе программы смотрят текущий каталог.
Сравнение с системой:
    ls -al /dev
EOF
