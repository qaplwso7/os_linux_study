Практическая работа 11. Команды для терминала Linux.
Файл в кодировке UTF-8. Открывай его в Cursor и копируй блоки отсюда.

Как вставлять:
1. Работай в терминале виртуальной машины Linux, в одном окне, не в PowerShell.
2. Копируй блок целиком, от первой команды до последней, и вставляй в терминал.
3. Блоки cat ... EOF тоже вставляются целиком: они сами создают файлы .c.
4. Иди по номерам. Следующий блок — только если предыдущий закончился без ошибки.
5. Системный диск не открывай. Все команды fdisk ниже относятся к disk.img.

Если fdisk задаст лишний вопрос, не подбирай ответы. Остановись и пришли скриншот.

========== 1. Какие диски уже есть ==========

sudo fdisk -l

Снимок этого вывода нужен в отчёт.
В жёлтые строки пункта 1 впиши число дисков, размеры и разделы.
/dev/loop от snap и /dev/ram в этот ответ не входят.

========== 2. Образ 512 МиБ и новый раздел ==========

mkdir -p "$HOME/os_pr11"
dd if=/dev/zero of="$HOME/os_pr11/disk.img" bs=1M count=512 status=progress
sudo -v
DEV=$(sudo losetup -fP --show "$HOME/os_pr11/disk.img")
echo "$DEV" > "$HOME/os_pr11/device.txt"
echo "Устройство: $DEV"
printf 'o\nn\np\n1\n\n\nt\n83\nw\n' | sudo fdisk "$DEV"
if command -v partprobe >/dev/null; then
  sudo partprobe "$DEV" || true
else
  echo "partprobe нет. Если понадобится: sudo apt-get install -y parted"
fi
sudo losetup -d "$DEV"
DEV=$(sudo losetup -fP --show "$HOME/os_pr11/disk.img")
echo "$DEV" > "$HOME/os_pr11/device.txt"
if [ -b "${DEV}p1" ]; then
  echo "Раздел: ${DEV}p1"
elif [ -b "${DEV}1" ]; then
  echo "Раздел: ${DEV}1"
else
  echo "Раздел не появился. Пришли скриншот, дальше пока не идём."
fi
lsblk "$DEV"

Что означают буквы, которые уходят в fdisk:
o — новая таблица DOS, n — новый раздел, p — основной, 1 — номер,
два пустых ответа — начало и конец по умолчанию, t и 83 — тип Linux, w — записать.

========== 3. Форматирование ext4 и монтирование ==========

DEV=$(cat "$HOME/os_pr11/device.txt")
if [ -b "${DEV}p1" ]; then PART="${DEV}p1"; else PART="${DEV}1"; fi
echo "Форматируем $PART"
sudo mkfs.ext4 -F "$PART"
sudo mkdir -p /mnt/mydisk
sudo mount "$PART" /mnt/mydisk
sudo chown "$USER:$USER" /mnt/mydisk
df -h /mnt/mydisk
findmnt /mnt/mydisk

========== 4. Удалить раздел и создать его заново на 256 МиБ ==========

cd "$HOME/os_pr11"
sudo umount /mnt/mydisk
DEV=$(cat "$HOME/os_pr11/device.txt")
printf 'd\nn\np\n1\n\n+256M\nt\n83\nw\n' | sudo fdisk "$DEV"
if command -v partprobe >/dev/null; then
  sudo partprobe "$DEV" || true
fi
sudo losetup -d "$DEV"
DEV=$(sudo losetup -fP --show "$HOME/os_pr11/disk.img")
echo "$DEV" > "$HOME/os_pr11/device.txt"
if [ -b "${DEV}p1" ]; then PART="${DEV}p1"; else PART="${DEV}1"; fi
echo "Новый раздел: $PART"
sudo mkfs.ext4 -F "$PART"
sudo mount "$PART" /mnt/mydisk
sudo chown "$USER:$USER" /mnt/mydisk
lsblk "$DEV"
df -h /mnt/mydisk

Буквы fdisk здесь: d — удалить единственный раздел, n p 1 — новый основной,
пустой ответ — начало по умолчанию, +256M — размер, t 83 — тип Linux, w — записать.

========== 5. df, du и каталог, который будем копировать ==========

mkdir -p "$HOME/os_pr11/source/docs"
printf 'Практическая работа 11\n' > "$HOME/os_pr11/source/readme.txt"
printf 'test\n' > "$HOME/os_pr11/source/docs/note.txt"
df -h
du -sh "$HOME/os_pr11/source"

В пункт 5 отчёта: строка df с самым большим Used и строка с самым маленьким Avail.
Туда же — объём, который напечатал du -sh.

========== 6. Строка в /etc/fstab ==========

Сначала проверка без перезагрузки. reboot — это следующий, отдельный блок.

cd "$HOME"
sudo umount /mnt/mydisk
DEV=$(cat "$HOME/os_pr11/device.txt")
sudo losetup -d "$DEV" || true
IMG="$HOME/os_pr11/disk.img"
START=$(LC_ALL=C sudo fdisk -l "$IMG" | awk '/Linux/ {print $2; exit}')
OFFSET=$((START * 512))
echo "start=$START offset=$OFFSET"
LINE="$IMG /mnt/mydisk ext4 loop,offset=${OFFSET},nofail 0 2"
echo "$LINE"
if [ ! -f /etc/fstab.bak.pr11 ]; then
  sudo cp /etc/fstab /etc/fstab.bak.pr11
fi
grep -q "$IMG" /etc/fstab || echo "$LINE" | sudo tee -a /etc/fstab >/dev/null
echo "----- строка в fstab -----"
grep "$IMG" /etc/fstab
sudo mount -a
findmnt /mnt/mydisk
df -h /mnt/mydisk

Если findmnt показывает /mnt/mydisk, можно перезагружаться.
Если mount -a пишет ошибку, перезагрузку не делай и пришли скриншот.

========== 6б. Перезагрузка, только после успешного mount -a ==========

sudo reboot

После входа в систему снова проверь:

findmnt /mnt/mydisk
df -h /mnt/mydisk

========== 7. Файл backup.c ==========

Перейди в каталог, где удобно хранить программы, например:

cd "$HOME/os_pr11"

Дальше вставь блок от cat до EOF включительно.

cat > backup.c << 'EOF'
#include <stdio.h>
#include <stdlib.h>

void check_free_space(void) {
    printf("Проверка свободного места на всех разделах:\n");
    system("df -h");
}

void calculate_directory_size(const char *directory) {
    char command[512];
    printf("Вычисление размера каталога: %s\n", directory);
    snprintf(command, sizeof(command), "du -sh \"%s\"", directory);
    system(command);
}

void list_block_devices(void) {
    printf("Список доступных разделов:\n");
    system("lsblk");
}

void backup_directory(const char *source, const char *destination) {
    char command[512];
    int result;

    printf("Начало резервного копирования каталога %s в %s\n", source, destination);
    snprintf(command, sizeof(command), "cp -r \"%s\" \"%s\"", source, destination);
    result = system(command);

    if (result == 0) {
        printf("Резервное копирование завершено успешно.\n");
    } else {
        printf("Ошибка при резервном копировании.\n");
    }
}

int main(void) {
    char source_directory[512];
    const char *destination_directory = "/mnt/mydisk";
    const char *home = getenv("HOME");

    if (home == NULL) {
        fprintf(stderr, "Не задана переменная HOME.\n");
        return 1;
    }

    snprintf(source_directory, sizeof(source_directory), "%s/os_pr11/source", home);

    check_free_space();
    calculate_directory_size(source_directory);
    list_block_devices();
    backup_directory(source_directory, destination_directory);

    return 0;
}
EOF

========== 8. Сборка и запуск backup ==========

cd "$HOME/os_pr11"
gcc backup.c -o backup
./backup
findmnt /mnt/mydisk
ls -la /mnt/mydisk/source

========== 9. Файл backup_auto.c ==========

cd "$HOME/os_pr11"

cat > backup_auto.c << 'EOF'
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <unistd.h>
#include <sys/wait.h>

#define MAX_FS 64

typedef struct {
    char device[256];
    char mount[256];
    unsigned long long avail;
} FsInfo;

static int path_ok(const char *s) {
    size_t i, n;

    if (s == NULL || s[0] != '/') {
        return 0;
    }
    n = strlen(s);
    if (n == 0 || n > 400) {
        return 0;
    }
    for (i = 0; i < n; i++) {
        unsigned char c = (unsigned char)s[i];
        if (isalnum(c) || c == '/' || c == '_' || c == '-' || c == '.') {
            continue;
        }
        return 0;
    }
    return 1;
}

static int command_ok(const char *cmd) {
    int status = system(cmd);
    if (status == -1 || !WIFEXITED(status)) {
        return 0;
    }
    return WEXITSTATUS(status) == 0;
}

static int load_filesystems(FsInfo *list, int max_count, int *count) {
    FILE *fp;
    char line[1024];

    *count = 0;
    fp = popen("LC_ALL=C df -B1 -P", "r");
    if (fp == NULL) {
        perror("df");
        return -1;
    }
    if (fgets(line, sizeof(line), fp) == NULL) {
        pclose(fp);
        return -1;
    }
    while (fgets(line, sizeof(line), fp) != NULL && *count < max_count) {
        FsInfo fs;
        char cap[32];
        unsigned long long blocks, used;

        if (sscanf(line, "%255s %llu %llu %llu %31s %255s",
                   fs.device, &blocks, &used, &fs.avail, cap, fs.mount) != 6) {
            continue;
        }
        list[*count] = fs;
        (*count)++;
    }
    pclose(fp);
    return 0;
}

static int mount_covers(const char *path, const char *mount) {
    size_t n = strlen(mount);

    if (strcmp(mount, "/") == 0) {
        return 1;
    }
    if (strncmp(path, mount, n) != 0) {
        return 0;
    }
    return path[n] == '\0' || path[n] == '/';
}

static int skip_candidate(const FsInfo *fs) {
    if (strncmp(fs->device, "/dev/", 5) != 0) {
        return 1;
    }
    if (strcmp(fs->mount, "/dev") == 0 || strncmp(fs->mount, "/dev/", 5) == 0) {
        return 1;
    }
    if (strncmp(fs->mount, "/proc", 5) == 0 || strncmp(fs->mount, "/sys", 4) == 0) {
        return 1;
    }
    if (strncmp(fs->mount, "/run", 4) == 0 || strncmp(fs->mount, "/snap", 5) == 0) {
        return 1;
    }
    return 0;
}

static unsigned long long directory_bytes(const char *directory) {
    char cmd[512];
    char line[256];
    FILE *fp;
    unsigned long long size = 0;

    snprintf(cmd, sizeof(cmd), "LC_ALL=C du -sb \"%s\"", directory);
    fp = popen(cmd, "r");
    if (fp == NULL) {
        return 0;
    }
    if (fgets(line, sizeof(line), fp) != NULL) {
        sscanf(line, "%llu", &size);
    }
    pclose(fp);
    return size;
}

static void show_info(const char *directory) {
    char cmd[512];

    printf("Проверка свободного места на всех разделах:\n");
    system("df -h");
    printf("\nСписок доступных разделов:\n");
    system("lsblk -o NAME,SIZE,TYPE,FSTYPE,MOUNTPOINT");
    printf("\nВычисление размера каталога: %s\n", directory);
    snprintf(cmd, sizeof(cmd), "du -sh \"%s\"", directory);
    system(cmd);
}

static int choose_target(const char *source, unsigned long long need, FsInfo *best) {
    FsInfo list[MAX_FS];
    int count = 0;
    int i, found = 0;
    char source_mount[256];

    if (load_filesystems(list, MAX_FS, &count) != 0) {
        return 0;
    }

    strcpy(source_mount, "/");
    for (i = 0; i < count; i++) {
        if (mount_covers(source, list[i].mount) &&
            strlen(list[i].mount) > strlen(source_mount)) {
            strncpy(source_mount, list[i].mount, sizeof(source_mount) - 1);
            source_mount[sizeof(source_mount) - 1] = '\0';
        }
    }

    printf("\nКаталог-источник находится на %s.\n", source_mount);
    printf("Нужно свободного места, с запасом 1 МиБ: %llu байт.\n", need);

    best->avail = 0;
    for (i = 0; i < count; i++) {
        if (skip_candidate(&list[i])) {
            continue;
        }
        if (strcmp(list[i].mount, source_mount) == 0) {
            continue;
        }
        if (access(list[i].mount, W_OK) != 0) {
            printf("Пропуск %s (%s): нет права на запись.\n",
                   list[i].device, list[i].mount);
            continue;
        }
        if (list[i].avail < need) {
            printf("Пропуск %s (%s): свободно %llu байт, этого мало.\n",
                   list[i].device, list[i].mount, list[i].avail);
            continue;
        }
        printf("Кандидат %s, точка %s, свободно %llu байт.\n",
               list[i].device, list[i].mount, list[i].avail);
        if (!found || list[i].avail > best->avail) {
            *best = list[i];
            found = 1;
        }
    }
    return found;
}

static void write_report(const char *path, const char *source, const char *dest_dir,
                         const FsInfo *best, unsigned long long need, int verified) {
    FILE *fp = fopen(path, "w");

    if (fp == NULL) {
        perror("Отчёт");
        return;
    }
    fprintf(fp, "Отчёт о резервном копировании\n");
    fprintf(fp, "Источник: %s\n", source);
    fprintf(fp, "Размер источника: %llu байт\n", need);
    fprintf(fp, "Раздел: %s\n", best->device);
    fprintf(fp, "Точка монтирования: %s\n", best->mount);
    fprintf(fp, "Свободно на разделе: %llu байт\n", best->avail);
    fprintf(fp, "Каталог копии: %s\n", dest_dir);
    fprintf(fp, "Сверка diff -rq: %s\n",
            verified ? "отличий нет" : "есть отличия или сверка не выполнена");
    fclose(fp);
    printf("Отчёт записан: %s\n", path);
}

static int default_source(char *buf, size_t n) {
    const char *home = getenv("HOME");

    if (home == NULL) {
        return -1;
    }
    if (snprintf(buf, n, "%s/os_pr11/source", home) >= (int)n) {
        return -1;
    }
    return 0;
}

int main(int argc, char **argv) {
    char source_buf[512];
    const char *source;
    FsInfo best;
    unsigned long long need;
    char dest_dir[512];
    char report_path[512];
    char cmd[1200];
    const char *base;
    int copied, verified;

    if (argc >= 2) {
        source = argv[1];
        if (!path_ok(source)) {
            fprintf(stderr, "Некорректный путь к каталогу.\n");
            return 1;
        }
    } else {
        if (default_source(source_buf, sizeof(source_buf)) != 0) {
            fprintf(stderr, "Не задана переменная HOME.\n");
            return 1;
        }
        source = source_buf;
    }

    if (access(source, R_OK) != 0) {
        perror(source);
        return 1;
    }

    show_info(source);
    need = directory_bytes(source);
    if (need == 0) {
        fprintf(stderr, "Не удалось определить размер каталога.\n");
        return 1;
    }

    if (!choose_target(source, need + 1048576ULL, &best)) {
        printf("Подходящий раздел не найден. Копирование не выполняется.\n");
        return 1;
    }

    printf("\nВыбран раздел %s (%s), свободно %llu байт.\n",
           best.device, best.mount, best.avail);

    base = strrchr(source, '/');
    base = (base == NULL) ? source : base + 1;
    if (snprintf(dest_dir, sizeof(dest_dir), "%s/%s", best.mount, base) >= (int)sizeof(dest_dir)) {
        fprintf(stderr, "Слишком длинный путь назначения.\n");
        return 1;
    }
    if (snprintf(report_path, sizeof(report_path), "%s/backup_report.txt", best.mount)
        >= (int)sizeof(report_path)) {
        fprintf(stderr, "Слишком длинный путь отчёта.\n");
        return 1;
    }

    printf("Начало резервного копирования каталога %s в %s\n", source, best.mount);
    if (snprintf(cmd, sizeof(cmd), "cp -r \"%s\" \"%s\"", source, best.mount) >= (int)sizeof(cmd)) {
        fprintf(stderr, "Слишком длинная команда копирования.\n");
        return 1;
    }
    copied = command_ok(cmd);
    if (!copied) {
        printf("Ошибка при резервном копировании.\n");
        write_report(report_path, source, dest_dir, &best, need, 0);
        return 1;
    }

    printf("Проверка копии командой diff -rq.\n");
    if (snprintf(cmd, sizeof(cmd), "diff -rq \"%s\" \"%s\"", source, dest_dir) >= (int)sizeof(cmd)) {
        fprintf(stderr, "Слишком длинная команда сверки.\n");
        return 1;
    }
    verified = command_ok(cmd);
    if (verified) {
        printf("Резервное копирование завершено успешно, файлы совпадают.\n");
    } else {
        printf("Копирование выполнено, но сверка обнаружила отличия.\n");
    }
    write_report(report_path, source, dest_dir, &best, need, verified);
    return verified ? 0 : 1;
}
EOF

========== 9б. Сборка и запуск модернизированной программы ==========

cd "$HOME/os_pr11"
gcc backup_auto.c -o backup_auto
./backup_auto
echo "----- отчёт -----"
cat /mnt/mydisk/backup_report.txt

Если программа выбрала не /mnt/mydisk, а другой раздел с большим свободным местом,
так и задумано: берётся максимум. В отчёт впиши тот раздел, который она напечатала.
Файл backup_report.txt тогда лежит в корне выбранной точки монтирования.
